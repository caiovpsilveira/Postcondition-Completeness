# count number of mutants generated per test
for file in $(ls FormalBench_mutation_analysis/completeness_results/hard/); do echo -n "$file "; ls FormalBench_mutation_analysis/completeness_results/hard/$file/mutants | wc -w; done					

# find all mutants that were accepted - had '0' verification failures (or find all tests that had all mutants rejected: add '-L' to grep)	
for file in $(find FormalBench_mutation_analysis/completeness_results/ -type f -name "completeness.txt"); do grep -H -P "\d: 0" $file; done
